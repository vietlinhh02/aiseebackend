```go
package reminder

import (
	"context"
	"fmt"
	"log/slog"
	"sync"
	"time"

	"github.com/redis/go-redis/v9"
)

type Reminder struct {
	Key   string
	DueAt time.Time
}

type Handler interface {
	LoadAll(ctx context.Context) ([]Reminder, error)
	OnDue(ctx context.Context, key string) error
}

type handlerEntry struct {
	handler  Handler
	redisKey string
}

type Scheduler struct {
	rdb      *redis.Client
	entries  []handlerEntry
	interval time.Duration
	mu       sync.Mutex
	cancel   context.CancelFunc
}

func New(rdb *redis.Client, interval time.Duration) *Scheduler {
	return &Scheduler{
		rdb:      rdb,
		interval: interval,
	}
}

func (s *Scheduler) Register(name string, h Handler) {
	s.entries = append(s.entries, handlerEntry{
		handler:  h,
		redisKey: "reminders:" + name,
	})
}

func (s *Scheduler) Start(ctx context.Context) error {
	ctx, s.cancel = context.WithCancel(ctx)

	for _, entry := range s.entries {
		reminders, err := entry.handler.LoadAll(ctx)
		if err != nil {
			return fmt.Errorf(
				"load reminders for %s: %w",
				entry.redisKey, err,
			)
		}
		for _, r := range reminders {
			s.rdb.ZAdd(ctx, entry.redisKey, redis.Z{
				Score:  float64(r.DueAt.Unix()),
				Member: r.Key,
			})
		}
		slog.Info("loaded reminders",
			"key", entry.redisKey,
			"count", len(reminders),
		)
	}

	ticker := time.NewTicker(s.interval)
	defer ticker.Stop()

	for {
		select {
		case <-ctx.Done():
			return nil
		case <-ticker.C:
			s.tick(ctx)
		}
	}
}

func (s *Scheduler) Stop() {
	if s.cancel != nil {
		s.cancel()
	}
}

func (s *Scheduler) tick(ctx context.Context) {
	now := float64(time.Now().Unix())
	for _, entry := range s.entries {
		members, err := s.rdb.ZRangeByScore(
			ctx, entry.redisKey,
			&redis.ZRangeBy{
				Min: "-inf",
				Max: fmt.Sprintf("%f", now),
			},
		).Result()
		if err != nil {
			slog.Error("redis zrangebyscore failed",
				"key", entry.redisKey, "error", err,
			)
			continue
		}
		for _, member := range members {
			s.rdb.ZRem(ctx, entry.redisKey, member)
			if err := entry.handler.OnDue(ctx, member); err != nil {
				slog.Error("reminder handler failed",
					"key", entry.redisKey,
					"member", member,
					"error", err,
				)
			}
		}
	}
}

func (s *Scheduler) Schedule(
	ctx context.Context, name string, r Reminder,
) error {
	key := "reminders:" + name
	return s.rdb.ZAdd(ctx, key, redis.Z{
		Score:  float64(r.DueAt.Unix()),
		Member: r.Key,
	}).Err()
}

func (s *Scheduler) Unschedule(
	ctx context.Context, name string, memberKey string,
) error {
	key := "reminders:" + name
	return s.rdb.ZRem(ctx, key, memberKey).Err()
}
```

package reminder

import (
	"context"
	"sync"
	"testing"
	"time"

	"github.com/redis/go-redis/v9"
)

type mockHandler struct {
	mu        sync.Mutex
	reminders []Reminder
	dueCalls  []string
}

func (m *mockHandler) LoadAll(
	_ context.Context,
) ([]Reminder, error) {
	return m.reminders, nil
}

func (m *mockHandler) OnDue(
	_ context.Context, key string,
) error {
	m.mu.Lock()
	defer m.mu.Unlock()
	m.dueCalls = append(m.dueCalls, key)
	return nil
}

func (m *mockHandler) getDueCalls() []string {
	m.mu.Lock()
	defer m.mu.Unlock()
	cp := make([]string, len(m.dueCalls))
	copy(cp, m.dueCalls)
	return cp
}

func newTestRedis(t *testing.T) *redis.Client {
	t.Helper()
	rdb := redis.NewClient(&redis.Options{
		Addr: "localhost:6379",
		DB:   15, // use DB 15 for tests
	})
	t.Cleanup(func() {
		rdb.FlushDB(context.Background())
		rdb.Close()
	})
	if err := rdb.Ping(context.Background()).Err(); err != nil {
		t.Skip("redis not available:", err)
	}
	rdb.FlushDB(context.Background())
	return rdb
}

func TestScheduler_FiresDueReminders(t *testing.T) {
	rdb := newTestRedis(t)
	handler := &mockHandler{
		reminders: []Reminder{
			{Key: "med-1:0", DueAt: time.Now().Add(-1 * time.Second)},
			{Key: "med-2:0", DueAt: time.Now().Add(1 * time.Hour)},
		},
	}

	s := New(rdb, 100*time.Millisecond)
	s.Register("test", handler)

	ctx, cancel := context.WithCancel(context.Background())
	go s.Start(ctx)

	time.Sleep(300 * time.Millisecond)
	cancel()

	calls := handler.getDueCalls()
	if len(calls) != 1 {
		t.Fatalf("expected 1 due call, got %d: %v", len(calls), calls)
	}
	if calls[0] != "med-1:0" {
		t.Errorf("expected med-1:0, got %s", calls[0])
	}
}

func TestScheduler_ScheduleAndUnschedule(t *testing.T) {
	rdb := newTestRedis(t)
	s := New(rdb, 100*time.Millisecond)
	handler := &mockHandler{}
	s.Register("test", handler)

	ctx := context.Background()

	err := s.Schedule(ctx, "test", Reminder{
		Key:   "apt-1:1h",
		DueAt: time.Now().Add(-1 * time.Second),
	})
	if err != nil {
		t.Fatal(err)
	}

	err = s.Unschedule(ctx, "test", "apt-1:1h")
	if err != nil {
		t.Fatal(err)
	}

	count, _ := rdb.ZCard(ctx, "reminders:test").Result()
	if count != 0 {
		t.Errorf("expected 0 members, got %d", count)
	}
}

package health

import (
	"strings"
	"unicode"

	"golang.org/x/text/unicode/norm"
)

var SymptomCategories = map[string]string{
	"dau dau":   "dau_dau",
	"chong mat": "chong_mat",
	"met moi":   "met_moi",
	"kho tho":   "kho_tho",
	"buon non":  "buon_non",
	"dau nguc":  "dau_nguc",
	"sot":       "sot",
	"ho":        "ho",
	"dau bung":  "dau_bung",
	"mo mat":    "mo_mat",
}

var MetricTypes = map[string]string{
	"huyet ap":    "huyet_ap",
	"duong huyet": "duong_huyet",
	"nhiet do":    "nhiet_do",
	"can nang":    "can_nang",
	"nhip tim":    "nhip_tim",
	"spo2":        "spo2",
}

var MetricUnits = map[string]string{
	"huyet_ap":    "mmHg",
	"duong_huyet": "mg/dL",
	"nhiet_do":    "°C",
	"can_nang":    "kg",
	"nhip_tim":    "bpm",
	"spo2":        "%",
}

func normalizeVietnamese(s string) string {
	s = strings.ToLower(s)
	t := norm.NFD.String(s)
	var b strings.Builder
	for _, r := range t {
		if !unicode.Is(unicode.Mn, r) {
			if r == 'đ' {
				b.WriteRune('d')
			} else {
				b.WriteRune(r)
			}
		}
	}
	return b.String()
}

func MatchSymptomCategory(input string) (string, bool) {
	normalized := normalizeVietnamese(input)
	for pattern, category := range SymptomCategories {
		if strings.Contains(normalized, pattern) {
			return category, true
		}
	}
	return "", false
}

func MatchMetricType(input string) (string, string, bool) {
	normalized := normalizeVietnamese(input)
	for pattern, metricType := range MetricTypes {
		if strings.Contains(normalized, pattern) {
			unit := MetricUnits[metricType]
			return metricType, unit, true
		}
	}
	return "", "", false
}

package health

import (
	"context"
	"fmt"

	"github.com/jackc/pgx/v5/pgtype"
	"github.com/team-dang/aiseebackend/internal/dbgen"
	"github.com/team-dang/aiseebackend/pkg/pgutil"
)

type Service struct {
	queries *dbgen.Queries
}

func NewService(queries *dbgen.Queries) *Service {
	return &Service{queries: queries}
}

func (s *Service) LogSymptom(
	ctx context.Context,
	userID, symptom, severity, notes string,
) (dbgen.HealthSymptom, error) {
	uid, err := pgutil.ParsePgUUID(userID)
	if err != nil {
		return dbgen.HealthSymptom{}, err
	}

	// Normalize symptom to predefined category if possible
	if cat, ok := MatchSymptomCategory(symptom); ok {
		symptom = cat
	}

	return s.queries.CreateHealthSymptom(
		ctx, dbgen.CreateHealthSymptomParams{
			UserID:   uid,
			Symptom:  symptom,
			Severity: pgtype.Text{String: severity, Valid: severity != ""},
			Notes:    pgtype.Text{String: notes, Valid: notes != ""},
		},
	)
}

func (s *Service) LogMetric(
	ctx context.Context,
	userID, metricType, value, unit string,
) (dbgen.HealthMetric, error) {
	uid, err := pgutil.ParsePgUUID(userID)
	if err != nil {
		return dbgen.HealthMetric{}, err
	}

	// Normalize metric type and fill default unit
	if mt, defaultUnit, ok := MatchMetricType(metricType); ok {
		metricType = mt
		if unit == "" {
			unit = defaultUnit
		}
	}

	return s.queries.CreateHealthMetric(
		ctx, dbgen.CreateHealthMetricParams{
			UserID:     uid,
			MetricType: metricType,
			Value:      value,
			Unit:       pgtype.Text{String: unit, Valid: unit != ""},
		},
	)
}

func (s *Service) GetSymptoms(
	ctx context.Context,
	userID string, limit, offset int32,
) ([]dbgen.HealthSymptom, error) {
	uid, err := pgutil.ParsePgUUID(userID)
	if err != nil {
		return nil, err
	}
	return s.queries.GetHealthSymptomsByUser(
		ctx, dbgen.GetHealthSymptomsByUserParams{
			UserID: uid,
			Limit:  limit,
			Offset: offset,
		},
	)
}

func (s *Service) GetMetrics(
	ctx context.Context,
	userID, metricType string,
	limit, offset int32,
) ([]dbgen.HealthMetric, error) {
	uid, err := pgutil.ParsePgUUID(userID)
	if err != nil {
		return nil, err
	}

	if metricType != "" {
		if mt, _, ok := MatchMetricType(metricType); ok {
			metricType = mt
		}
		return s.queries.GetHealthMetricsByType(
			ctx, dbgen.GetHealthMetricsByTypeParams{
				UserID:     uid,
				MetricType: metricType,
				Limit:      limit,
				Offset:     offset,
			},
		)
	}

	return s.queries.GetHealthMetricsByUser(
		ctx, dbgen.GetHealthMetricsByUserParams{
			UserID: uid,
			Limit:  limit,
			Offset: offset,
		},
	)
}

func ValidateMetricValue(
	metricType, value string,
) error {
	if value == "" {
		return fmt.Errorf("value is required")
	}
	return nil
}

func (s *Service) OnDue(
	ctx context.Context, key string,

	// Parse "medication_id:time_index"
	parts := strings.SplitN(key, ":", 2)
	if len(parts) != 2 {
		return fmt.Errorf("invalid reminder key: %s", key)
	}
	medID := parts[0]
	timeIdx := parts[1]

	medUUID, err := pgutil.ParsePgUUID(medID)
	if err != nil {
		return err
	}

	// Get all medications matching this ID
	// (MVP: linear scan is fine for small user counts)
	meds, err := s.queries.GetActiveMedicationsAll(ctx)
	if err != nil {
		return err
	}

	var med *dbgen.Medication
	for _, m := range meds {
		if pgutil.PgUUIDToString(m.ID) == medID {
			med = &m
			break
		}
	}
	if med == nil {
		return nil // medication was deactivated
	}

	// Check if already taken today
	logs, _ := s.queries.GetMedicationLogToday(ctx, medUUID)
	alreadyTaken := false
	for _, log := range logs {
		if !log.Skipped.Bool {
			alreadyTaken = true
			break
		}
	}

	if !alreadyTaken {
		// Lookup user for push tokens
		user, userErr := s.queries.GetUserByID(ctx, med.UserID)
		if userErr == nil {
			s.pusher.Send(ctx, push.UserTokens{
				FcmToken:      user.FcmToken,
				ApnsToken:     user.ApnsToken,
				ApnsVoipToken: user.ApnsVoipToken,
			}, push.Notification{
				Title: "Nhac uong thuoc",
				Body: fmt.Sprintf(
					"Den gio uong %s %s", med.Name, med.Dosage,
				),
				Type: "normal",
				Data: map[string]string{
					"type":          "medication_reminder",
					"medication_id": medID,
				},
			})
		}
	}

	// Reschedule for tomorrow (daily only for MVP)
	var idx int
	fmt.Sscanf(timeIdx, "%d", &idx)
	if idx < len(med.Times) {
		dueAt, parseErr := nextOccurrence(med.Times[idx])
		if parseErr == nil {
			// Add 24h to get tomorrow
			dueAt = dueAt.Add(24 * time.Hour)
			s.scheduler.Schedule(ctx, "medications", reminder.Reminder{
				Key:   key,
				DueAt: dueAt,
			})
		}
	}

	return nil
}

func nextOccurrence(timeStr string) (time.Time, error) {
	// Parse "HH:MM" format
	loc, _ := time.LoadLocation("Asia/Ho_Chi_Minh")
	now := time.Now().In(loc)
	t, err := time.ParseInLocation("15:04", timeStr, loc)
	if err != nil {
		return time.Time{}, fmt.Errorf(
			"invalid time format %q: %w", timeStr, err,
		)
	}
	next := time.Date(
		now.Year(), now.Month(), now.Day(),
		t.Hour(), t.Minute(), 0, 0, loc,
	)
	if next.Before(now) {
		next = next.Add(24 * time.Hour)
	}
	return next, nil
}

func normalizeVN(s string) string {
	s = strings.ToLower(s)
	t := norm.NFD.String(s)
	var b strings.Builder
	for _, r := range t {
		if !unicode.Is(unicode.Mn, r) {
			if r == 'đ' {
				b.WriteRune('d')
			} else {
				b.WriteRune(r)
			}
		}
	}
	return b.String()
}

func (h *Handler) handleSetMedication(
	ctx context.Context, req ToolCallRequest,
) (any, error) {
	name, _ := req.Params["medicine_name"].(string)
	dosage, _ := req.Params["dosage"].(string)
	frequency, _ := req.Params["frequency"].(string)
	if name == "" || dosage == "" {
		return nil, fmt.Errorf(
			"medicine_name and dosage are required",
		)
	}
	if frequency == "" {
		frequency = "daily"
	}

	// Parse times array
	var times []string
	if rawTimes, ok := req.Params["times"].([]any); ok {
		for _, rt := range rawTimes {
			if t, ok := rt.(string); ok {
				times = append(times, t)
			}
		}
	}
	if len(times) == 0 {
		return nil, fmt.Errorf("times array is required")
	}

	med, err := h.medication.Create(
		ctx, req.UserID, name, dosage, times, frequency,
	)
	if err != nil {
		return nil, err
	}

	return map[string]any{
		"status":     "created",
		"medication": map[string]any{
			"name":  med.Name,
			"dosage": med.Dosage,
			"times": med.Times,

		},
	}, nil
}

	datetimeStr, _ := req.Params["datetime"].(string)
	location, _ := req.Params["location"].(string)
	notes, _ := req.Params["notes"].(string)
	if title == "" || datetimeStr == "" {
		return nil, fmt.Errorf(
			"title and datetime are required",
		)
	}

	loc, locErr := time.LoadLocation("Asia/Ho_Chi_Minh")
	if locErr != nil {
		return nil, fmt.Errorf("load timezone: %w", locErr)
	}
	datetime, err := time.ParseInLocation(
		"2006-01-02 15:04", datetimeStr, loc,
	)
	if err != nil {
		// Try ISO format
		datetime, err = time.ParseInLocation(
			time.RFC3339, datetimeStr, loc,
		)
		if err != nil {
			return nil, fmt.Errorf(
				"invalid datetime format, use YYYY-MM-DD HH:MM",
			)
		}
	}

	apt, err := h.appointment.Create(
		ctx, req.UserID, title, location, datetime, notes,
	)
	if err != nil {
		return nil, err
	}

	return map[string]any{
		"status": "created",
		"appointment": map[string]any{
			"title":    apt.Title,
			"datetime": apt.Datetime.Time.In(loc).Format(
				"2006-01-02 15:04",
			),
			"location": location,
		},
		"schedule_action": map[string]any{
			"action": "schedule_reminder",
			"params": map[string]any{
				"type":     "appointment",
				"title":    apt.Title,
				"datetime": datetimeStr,
			},
		},
	}, nil
