package pgutil

import (
	"fmt"

	"github.com/google/uuid"
	"github.com/jackc/pgx/v5/pgtype"
)

func ParsePgUUID(s string) (pgtype.UUID, error) {
	uid, err := uuid.Parse(s)
	if err != nil {
		return pgtype.UUID{}, fmt.Errorf("invalid uuid: %w", err)
	}
	return pgtype.UUID{Bytes: uid, Valid: true}, nil
}

func PgUUIDToString(u pgtype.UUID) string {
	if !u.Valid {
		return ""
	}
	uid := uuid.UUID(u.Bytes)
	return uid.String()
}
```

package push

import (
	"context"
	"log/slog"

	"github.com/jackc/pgx/v5/pgtype"
)

// UserTokens holds the push token fields needed for sending.
// Matches the token fields from dbgen.User without importing dbgen.
type UserTokens struct {
	FcmToken      pgtype.Text
	ApnsToken     pgtype.Text
	ApnsVoipToken pgtype.Text
}

type Notification struct {
	Title string
	Body  string
	Type  string            // "normal" | "voip"
	Data  map[string]string // custom payload
}

type Sender interface {
	Send(ctx context.Context, token string, n Notification) error
	Name() string
}

type Pusher struct {
	fcm  Sender
	apns Sender
}

func NewPusher(fcm, apns Sender) *Pusher {
	return &Pusher{fcm: fcm, apns: apns}
}

func (p *Pusher) Send(ctx context.Context, tokens UserTokens, n Notification) {
	if tokens.FcmToken.Valid && tokens.FcmToken.String != "" {
		if err := p.fcm.Send(ctx, tokens.FcmToken.String, n); err != nil {
			slog.Error("fcm push failed", "error", err)
		}
	}

	if n.Type == "voip" && tokens.ApnsVoipToken.Valid && tokens.ApnsVoipToken.String != "" {
		if err := p.apns.Send(ctx, tokens.ApnsVoipToken.String, n); err != nil {
			slog.Error("apns voip push failed", "error", err)
		}
	} else if tokens.ApnsToken.Valid && tokens.ApnsToken.String != "" {
		if err := p.apns.Send(ctx, tokens.ApnsToken.String, n); err != nil {
			slog.Error("apns push failed", "error", err)
		}
	}
}

package push

import (
	"bytes"
	"context"
	"encoding/json"
	"fmt"
	"net/http"
	"os"
	"sync"
	"time"

	"golang.org/x/oauth2/google"
)

type FCMSender struct {
	projectID          string
	serviceAccountPath string
	httpClient         *http.Client
	tokenSource        struct {
		sync.Mutex
		token   string
		expires time.Time
	}
}

func NewFCMSender(serviceAccountPath string) (*FCMSender, error) {
	data, err := os.ReadFile(serviceAccountPath)
	if err != nil {
		return nil, fmt.Errorf("read service account: %w", err)
	}

	var sa struct {
		ProjectID string `json:"project_id"`
	}
	if err := json.Unmarshal(data, &sa); err != nil {
		return nil, fmt.Errorf("parse service account: %w", err)
	}

	return &FCMSender{
		projectID:          sa.ProjectID,
		serviceAccountPath: serviceAccountPath,
		httpClient:         &http.Client{Timeout: 10 * time.Second},
	}, nil
}

func (f *FCMSender) Name() string { return "fcm" }

func (f *FCMSender) getAccessToken(ctx context.Context) (string, error) {
	f.tokenSource.Lock()
	defer f.tokenSource.Unlock()

	if f.tokenSource.token != "" && time.Now().Before(f.tokenSource.expires) {
		return f.tokenSource.token, nil
	}

	data, err := os.ReadFile(f.serviceAccountPath)
	if err != nil {
		return "", err
	}

	creds, err := google.CredentialsFromJSON(ctx, data, "https://www.googleapis.com/auth/firebase.messaging")
	if err != nil {
		return "", err
	}

	tok, err := creds.TokenSource.Token()
	if err != nil {
		return "", err
	}

	f.tokenSource.token = tok.AccessToken
	f.tokenSource.expires = tok.Expiry.Add(-time.Minute)
	return tok.AccessToken, nil
}

func (f *FCMSender) Send(ctx context.Context, token string, n Notification) error {
	accessToken, err := f.getAccessToken(ctx)
	if err != nil {
		return fmt.Errorf("get access token: %w", err)
	}

	msg := map[string]any{
		"message": map[string]any{
			"token": token,
			"data":  n.Data,
		},
	}

	// For normal notifications, include notification block
	// For incoming calls (voip), use data-only so app handles display
	if n.Type != "voip" {
		msg["message"].(map[string]any)["notification"] = map[string]string{
			"title": n.Title,
			"body":  n.Body,
		}
	}

	// Android high priority for calls
	if n.Type == "voip" {
		msg["message"].(map[string]any)["android"] = map[string]string{
			"priority": "HIGH",
		}
	}

	body, err := json.Marshal(msg)
	if err != nil {
		return err
	}

	url := fmt.Sprintf("https://fcm.googleapis.com/v1/projects/%s/messages:send", f.projectID)
	req, err := http.NewRequestWithContext(ctx, "POST", url, bytes.NewReader(body))
	if err != nil {
		return err
	}
	req.Header.Set("Authorization", "Bearer "+accessToken)
	req.Header.Set("Content-Type", "application/json")

	resp, err := f.httpClient.Do(req)
	if err != nil {
		return fmt.Errorf("fcm request: %w", err)
	}
	defer resp.Body.Close()

	if resp.StatusCode != http.StatusOK {
		return fmt.Errorf("fcm response: %d", resp.StatusCode)
	}

	return nil
}

package push

import (
	"context"
	"encoding/json"
	"fmt"

	"github.com/sideshow/apns2"
	"github.com/sideshow/apns2/token"
)

type APNsSender struct {
	client   *apns2.Client
	bundleID string
}

func NewAPNsSender(keyPath, keyID, teamID, bundleID string, production bool) (*APNsSender, error) {
	authKey, err := token.AuthKeyFromFile(keyPath)
	if err != nil {
		return nil, fmt.Errorf("load apns key: %w", err)
	}

	tok := &token.Token{
		AuthKey: authKey,
		KeyID:   keyID,
		TeamID:  teamID,
	}

	client := apns2.NewTokenClient(tok)
	if production {
		client = client.Production()
	} else {
		client = client.Development()
	}

	return &APNsSender{client: client, bundleID: bundleID}, nil
}

func (a *APNsSender) Name() string { return "apns" }

func (a *APNsSender) Send(_ context.Context, deviceToken string, n Notification) error {
	payload := map[string]any{}

	if n.Type == "voip" {
		payload["aps"] = map[string]any{}
		for k, v := range n.Data {
			payload[k] = v
		}
	} else {
		payload["aps"] = map[string]any{
			"alert": map[string]string{
				"title": n.Title,
				"body":  n.Body,
			},
			"sound": "default",
		}
		if n.Data != nil {
			payload["data"] = n.Data
		}
	}

	payloadBytes, err := json.Marshal(payload)
	if err != nil {
		return err
	}

	notification := &apns2.Notification{
		DeviceToken: deviceToken,
		Payload:     payloadBytes,
		Priority:    apns2.PriorityHigh,
	}

	if n.Type == "voip" {
		notification.Topic = a.bundleID + ".voip"
		notification.PushType = apns2.PushTypeVoIP
	} else {
		notification.Topic = a.bundleID
	}

	resp, err := a.client.Push(notification)
	if err != nil {
		return fmt.Errorf("apns push: %w", err)
	}

	if !resp.Sent() {
		return fmt.Errorf("apns rejected: %s (%d)", resp.Reason, resp.StatusCode)
	}

	return nil
}

import (
	"strings"
	"unicode"

	"golang.org/x/text/unicode/norm"
)

// normalizeVietnamese lowercases and strips Vietnamese diacritics for fuzzy matching.
func normalizeVietnamese(s string) string {
	s = strings.ToLower(s)
	t := norm.NFD.String(s)
	var b strings.Builder
	for _, r := range t {
		if !unicode.Is(unicode.Mn, r) { // Mn = Mark, Nonspacing (diacritics)
			if r == 'đ' {
				b.WriteRune('d')
			} else {
				b.WriteRune(r)
			}
		}
	}
	return b.String()
}

func matchContact(query, displayName, relationship string) bool {
	q := normalizeVietnamese(query)
	return strings.Contains(normalizeVietnamese(displayName), q) ||
		strings.Contains(normalizeVietnamese(relationship), q)
}

type FamilyMember struct {
	User         dbgen.User
	Relationship string
}

func (s *Service) ResolveContact(ctx context.Context, userID, contactName string) (FamilyMember, error) {
	uid, err := pgutil.ParsePgUUID(userID)
	if err != nil {
		return FamilyMember{}, err
	}

	rows, err := s.queries.GetFamilyMembersWithDetails(ctx, uid)
	if err != nil {
		return FamilyMember{}, fmt.Errorf("get family members: %w", err)
	}

	var matches []FamilyMember
	for _, row := range rows {
		member := FamilyMember{
			User: dbgen.User{
				ID:            row.ID,
				GoogleID:      row.GoogleID,
				Email:         row.Email,
				DisplayName:   row.DisplayName,
				Role:          row.Role,
				IsVolunteer:   row.IsVolunteer,
				AvatarUrl:     row.AvatarUrl,
				FcmToken:      row.FcmToken,
				ApnsToken:     row.ApnsToken,
				ApnsVoipToken: row.ApnsVoipToken,
				Settings:      row.Settings,
				CreatedAt:     row.CreatedAt,
				UpdatedAt:     row.UpdatedAt,
			},
			Relationship: row.Relationship,
		}
		if matchContact(contactName, row.DisplayName, row.Relationship) {
			matches = append(matches, member)
		}
	}

	switch len(matches) {
	case 0:
		return FamilyMember{}, fmt.Errorf("contact not found: %s", contactName)
	case 1:
		return matches[0], nil
	default:

	}
}

package chat

import (
	"context"

	"log/slog"
	"sync"

	"nhooyr.io/websocket"
)

type Hub struct {
	mu    sync.RWMutex
	conns map[string]*websocket.Conn // userID → conn
}

func NewHub() *Hub {
	return &Hub{conns: make(map[string]*websocket.Conn)}
}

func (h *Hub) Add(userID string, conn *websocket.Conn) {
	h.mu.Lock()
	defer h.mu.Unlock()

	if old, ok := h.conns[userID]; ok {
		old.CloseNow()
	}
	h.conns[userID] = conn
}

func (h *Hub) Remove(userID string) {
	h.mu.Lock()
	defer h.mu.Unlock()
	delete(h.conns, userID)
}

type WSMessage struct {
	Type    string `json:"type"`
	Message any    `json:"message"`
}

func (h *Hub) SendToUser(userID string, msg WSMessage) {
	h.mu.RLock()
	conn, ok := h.conns[userID]
	h.mu.RUnlock()

	if !ok {
		return
	}

	data, err := json.Marshal(msg)
	if err != nil {
		slog.Error("marshal ws message", "error", err)
		return
	}

	if err := conn.Write(context.Background(), websocket.MessageText, data); err != nil {
		slog.Error("write ws message", "error", err, "user_id", userID)
		h.Remove(userID)
	}
}

func (h *Hub) IsOnline(userID string) bool {
	h.mu.RLock()
	defer h.mu.RUnlock()
	_, ok := h.conns[userID]
	return ok
}

func (h *Handler) SendMessage(w http.ResponseWriter, r *http.Request) {

	if req.ContentText == "" && req.VoiceURL == "" {

		return

	if err != nil {
		response.Err(w, http.StatusInternalServerError, "send_failed", "Failed to send message")
		return
	}

}

func (h *Handler) GetMessages(w http.ResponseWriter, r *http.Request) {
	userID := middleware.UserIDFromContext(r.Context())
	contactID := chi.URLParam(r, "contactID")

	linked, err := h.userService.CheckFamilyLink(r.Context(), userID, contactID)
	if err != nil || !linked {
		response.Err(w, http.StatusForbidden, "not_family", "You can only view family messages")
		return
	}

	limitStr := r.URL.Query().Get("limit")
	limit := int32(50)
	if limitStr != "" {
		l, err := strconv.Atoi(limitStr)
		if err == nil && l > 0 && l <= 100 {
			limit = int32(l)
		}
	}

	beforeID := r.URL.Query().Get("before")

	var msgs []dbgen.Message
	if beforeID != "" {
		msgs, err = h.service.GetMessagesBefore(r.Context(), userID, contactID, beforeID, limit)
	} else {
		msgs, err = h.service.GetMessages(r.Context(), userID, contactID, limit)
	}
	if err != nil {
		response.Err(w, http.StatusInternalServerError, "fetch_failed", "Failed to fetch messages")
		return
	}

func (h *Handler) WebSocket(w http.ResponseWriter, r *http.Request) {
	tokenStr := r.URL.Query().Get("token")
	if tokenStr == "" {
		response.Err(w, http.StatusUnauthorized, "missing_token", "token query param required")
		return
	}

	token, err := jwt.Parse(tokenStr, func(t *jwt.Token) (any, error) {
		return []byte(h.jwtSecret), nil
	})
	if err != nil || !token.Valid {
		response.Err(w, http.StatusUnauthorized, "invalid_token", "Invalid token")
		return
	}

	claims, ok := token.Claims.(jwt.MapClaims)
	if !ok {
		response.Err(w, http.StatusUnauthorized, "invalid_claims", "Invalid claims")
		return
	}
	userID, _ := claims["sub"].(string)

	conn, err := websocket.Accept(w, r, &websocket.AcceptOptions{
		InsecureSkipVerify: true, // Allow any origin for MVP
	})
	if err != nil {
		slog.Error("websocket accept failed", "error", err)
		return
	}

	h.hub.Add(userID, conn)
	slog.Info("websocket connected", "user_id", userID)

	// Use a background context for the long-lived connection
	// (r.Context() may be canceled after the HTTP upgrade completes)
	ctx := context.Background()
	for {
		_, _, err := conn.Read(ctx)
		if err != nil {
			break
		}
	}

	h.hub.Remove(userID)
	slog.Info("websocket disconnected", "user_id", userID)
}

		return
	}

	if err != nil || !linked {
		response.Err(w, http.StatusForbidden, "not_family", "You can only call family members")
		return
	}

	if err != nil {
		response.Err(w, http.StatusInternalServerError, "caller_not_found", "Caller not found")
		return

		return

"""LiveKit data channel bridge for agent → app communication."""

import asyncio
import json
import logging
import uuid

logger = logging.getLogger("aisee-agent")

TIMEOUT_SECONDS = 5
MAX_RETRIES = 1


class DataChannelTimeoutError(Exception):
    """Raised when app does not ACK within timeout."""


class DataChannelBridge:
    """Send tool actions to app via LiveKit data channel."""

    def __init__(self):
        self._pending: dict[str, asyncio.Future] = {}

    def on_data_received(self, data: bytes):
        """Handle incoming data channel messages from app.

        Call this from the LiveKit room data received callback.
        """
        try:
            msg = json.loads(data)
        except json.JSONDecodeError:
            logger.warning("invalid data channel message: %s", data)
            return

        if msg.get("type") != "tool_action_ack":
            return

        request_id = msg.get("request_id")
        if request_id in self._pending:
            self._pending[request_id].set_result(msg)

    async def send_action(
        self,
        local_participant,
        action: str,
        params: dict,
    ) -> dict:
        """Send a tool action and wait for ACK from the app."""
        request_id = str(uuid.uuid4())
        message = json.dumps({
            "type": "tool_action",
            "request_id": request_id,
            "action": action,
            "params": params,
        })

        for attempt in range(MAX_RETRIES + 1):
            future = asyncio.get_event_loop().create_future()
            self._pending[request_id] = future

            try:
                await local_participant.publish_data(
                    message.encode(),
                    reliable=True,
                )

                result = await asyncio.wait_for(
                    future, timeout=TIMEOUT_SECONDS,
                )

                if result.get("status") == "error":
                    error_msg = result.get(
                        "error_message", "Unknown error",
                    )
                    return {"status": "error", "error": error_msg}

                return {"status": "ok"}

            except asyncio.TimeoutError:
                if attempt < MAX_RETRIES:
                    logger.warning(
                        "data channel timeout, retrying "
                        "(attempt %d): %s",
                        attempt + 1,
                        action,
                    )
                    continue
                raise DataChannelTimeoutError(
                    f"App did not respond to {action} "
                    f"after {MAX_RETRIES + 1} attempts"
                )
            finally:
                self._pending.pop(request_id, None)
```

import logging

from livekit.agents import AgentSession, Agent, RoomInputOptions

from livekit.plugins import google

from backend import BackendClient
from config import GEMINI_MODEL, GO_BACKEND_URL, INTERNAL_API_KEY
from data_channel import DataChannelBridge
from tools import ALL_TOOLS, init_tools

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger("aisee-agent")

SYSTEM_PROMPT = """
Bạn là AiSee, trợ lý AI cho người suy giảm thị lực tại Việt Nam.

RESPOND IN VIETNAMESE. YOU MUST RESPOND UNMISTAKABLY IN VIETNAMESE.

Nguyên tắc:
- Luôn nói tiếng Việt, giọng thân thiện, rõ ràng, chậm vừa phải

Lời chào: Khi bắt đầu cuộc trò chuyện, hãy nói:
"Xin chào, tôi là AiSee. Tôi có thể giúp gì cho bạn?"

class AiSeeAgent(Agent):
    def __init__(self):
        super().__init__(
            instructions=SYSTEM_PROMPT,
        )


async def entrypoint(ctx):
    agent = AiSeeAgent()
    bridge = DataChannelBridge()
    backend = BackendClient(GO_BACKEND_URL, INTERNAL_API_KEY)

    # Get user_id from the first remote participant's identity
    user_id = ""
    for participant in ctx.room.remote_participants.values():
        user_id = participant.identity
        break

    if not user_id:
        logger.warning("No remote participant found, using empty user_id")

    # Wire data channel callbacks
    # Note: verify the exact callback signature for your livekit-agents version.
    # The data param may be a DataPacket object — use data.data for raw bytes.
    @ctx.room.on("data_received")
    def on_data(data_packet, *_args, **_kwargs):
        raw = data_packet.data if hasattr(data_packet, "data") else data_packet
        bridge.on_data_received(raw)

    # Initialize tool dependencies
    init_tools(
        backend=backend,
        bridge=bridge,
        local_participant=ctx.room.local_participant,
        user_id=user_id,
    )

    session = AgentSession(
        llm=google.realtime.RealtimeModel(
            model=GEMINI_MODEL,
        ),
    )

    for tool in ALL_TOOLS:
        session.register_tool(tool)

    await session.start(
        agent=agent,
        room=ctx.room,
        room_input_options=RoomInputOptions(video_enabled=False),
    )


if __name__ == "__main__":
    from livekit.agents import WorkerOptions, cli

    cli.run_app(
        WorkerOptions(
            entrypoint_fnc=entrypoint,
        )
    )
